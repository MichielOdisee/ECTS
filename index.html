Voorbeeld SPARQL Query

PREFIX eloict: <http://odisee.be/eloict#>
PREFIX rdfs:   <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?vak ?naam WHERE {
    ?vak a eloict:CourseUnit ;
         rdfs:label ?naam .
}
LIMIT 20
